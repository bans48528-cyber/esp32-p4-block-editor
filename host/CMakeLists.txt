cmake_minimum_required(VERSION 3.16)
project(editor_host C)
set(CMAKE_C_STANDARD 11)
set(LVGL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../C3986-新7.29JC4880P443C_I_W/JC4880P443C_I_W/1-Demo/idf_examples/ESP-IDF/lvgl_sw_rotation/components/lvgl__lvgl" CACHE PATH "Local vendor LVGL 9.2.2")
file(GLOB_RECURSE LV_SOURCES "${LVGL_DIR}/src/*.c")
add_library(lvgl STATIC ${LV_SOURCES})
target_include_directories(lvgl PUBLIC "${LVGL_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
  target_compile_options(lvgl PUBLIC /utf-8)
endif()
add_executable(editor_host test_editor.c ../main/editor.c ../main/projects.c ../main/model.c ../main/block_shape.c ../main/block_icons.c ../main/toolbar_icons.c ../main/board_font.c)
target_include_directories(editor_host PRIVATE ../main)
target_link_libraries(editor_host PRIVATE lvgl)
add_executable(editor_preview WIN32 window.c ../main/editor.c ../main/projects.c ../main/model.c ../main/block_shape.c ../main/block_icons.c ../main/toolbar_icons.c ../main/board_font.c)
target_include_directories(editor_preview PRIVATE ../main)
target_link_libraries(editor_preview PRIVATE lvgl user32 gdi32)

# Build another preview without closing a running window and losing its draft.
set(BOARD_PREVIEW_DIRECTORY "" CACHE PATH "Optional directory for an independent preview executable")
if(BOARD_PREVIEW_DIRECTORY)
  set_target_properties(editor_preview PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${BOARD_PREVIEW_DIRECTORY}")
endif()
